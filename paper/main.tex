% Float placement. LaTeX's defaults are conservative, and with this many
% figures and tables they overflow: a float that cannot be placed
% blocks every later float of its class, so the queue drains at the end of the
% document. elsarticle does not change these parameters, so the values below
% apply unchanged. Defaults are in the comments. `floatpagefraction' must stay
% below `topfraction' or LaTeX will build float-only pages in preference to
% placing floats on text pages. At 0.70 a table and a figure that together
% filled three quarters of a page were set as a float-only page with a quarter
% of it blank (the band table and spectrum figure of sections/mechanism.tex);
% at 0.80 they share pages with text. The appendices,
% whose generated tables are two thirds of a page tall and cannot share a page
% with text, set it back to 0.70 (sections/appendix.tex).
\renewcommand{\topfraction}{0.85}          % 0.7
\renewcommand{\bottomfraction}{0.70}       % 0.3
\renewcommand{\textfraction}{0.15}         % 0.2
\renewcommand{\floatpagefraction}{0.80}    % 0.5
\setcounter{topnumber}{4}                  % 2
\setcounter{bottomnumber}{3}               % 1
\setcounter{totalnumber}{6}                % 3

% A figure that degrades to a visible placeholder rather than a build failure,
% so the manuscript compiles on a machine that has the text but not the sweep
% output. The label is carried by both branches, so a missing figure does not
% turn every \ref to it into a dangling one. A vector copy (same name, .pdf)
% is preferred when present, as Elsevier's artwork guidelines ask; the raster
% named in the call is the fallback.
% #1 width fraction  #2 file  #3 caption  #4 label
\makeatletter
\newcommand{\guardedfigure}[4]{%
  \begin{figure}[tbp]\centering
  \filename@parse{#2}\edef\gf@vector{\filename@base.pdf}%
  \IfFileExists{figures/\gf@vector}{\includegraphics[width=#1\linewidth]{\gf@vector}}{%
  \IfFileExists{figures/#2}{\includegraphics[width=#1\linewidth]{#2}}{%
    \fbox{\parbox{0.8\linewidth}{\centering\vspace{6mm}
    % \detokenize so a filename containing an underscore prints as text
    % instead of failing the build with "Missing $ inserted".
    \textbf{\detokenize{figures/#2} not built}\\[1mm]
    run \texttt{awwl figures}\vspace{6mm}}}}}%
  \caption{#3}\label{#4}
  \end{figure}}

% Run-in paragraph headings. elsarticle appends a full stop to every
% \paragraph title, while the section files write their titles with their own
% punctuation ("\paragraph{Metric scope.}"), which printed "scope..". Add the
% stop only when the title does not already end in punctuation, judged by the
% space factor the last character left (the test amsthm uses).
\def\els@addstop{\ifhmode\ifnum\spacefactor>\@m\else.\fi\else.\fi}
\def\els@aparagraph[#1]#2{\elsparagraph[#1]{#2\els@addstop}}
\def\els@bparagraph#1{\elsparagraph*{#1\els@addstop}}

% Appendices. elsarticle's \appendix sets \thesection to "Appendix~A", which
% it prints in the heading and returns from every \ref, so text written as
% Appendix~\ref{app:theory} would read "Appendix Appendix A". Keep the
% "Appendix A." heading and make \ref return the letter alone (A, A.1, ...);
% the text therefore writes Appendix~\ref{app:...}.
% elsarticle restarts equation numbers in each appendix but lets tables and
% figures run on from the main text (Table B.14); Elsevier numbers them per
% appendix (Table A.1, B.1, ...). hyperref's patched \@addtoreset makes the
% anchors unique (table.B.1), so the restart does not clash with Table 1.
\newcommand{\startappendices}{%
  \appendix
  \setcounter{table}{0}\setcounter{figure}{0}%
  \@addtoreset{table}{section}\@addtoreset{figure}{section}%
  \renewcommand{\thesection}{\Alph{section}}%
  \def\@seccntformat##1{\csname app@prefix@##1\endcsname
    \csname the##1\endcsname\@seccntDot\hskip 0.5em}%
  \def\app@prefix@section{\appendixname}}
\makeatother
